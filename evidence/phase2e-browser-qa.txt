    await page.setViewportSize({width:target.width,height:900});
    await page.goto(`http://localhost:4173${target.path}`, {waitUntil:'networkidle'});
    const check = await page.evaluate(() => {
      const body=document.body, html=document.documentElement;
      const enabledCommerce=[...document.querySelectorAll('a,button,form,input,select,textarea')].filter(el=>{
        const txt=((el.innerText||el.value||el.getAttribute('aria-label')||'')+'').toLowerCase();
        const href=(el.getAttribute('href')||'').toLowerCase();
        return /(buy|purchase|checkout|pay now|add to cart|download now|get instant access)/.test(txt) || /(gumroad|checkout|stripe|paypal|lemonsqueezy|shopify)/.test(href);
      }).filter(el=>!(el.disabled || el.getAttribute('aria-disabled')==='true'));
      const scripts=[...document.scripts].map(s=>s.src||s.textContent||'').filter(v=>/gumroad|stripe|paypal|lemonsqueezy|shopify/i.test(v));
      const overflow=Math.max(0, document.documentElement.scrollWidth-window.innerWidth);
      const focusables=[...document.querySelectorAll('a[href],button:not([disabled]),input:not([disabled]),select:not([disabled]),textarea:not([disabled])')];
      const directions={html:html.getAttribute('dir')||'ltr',lang:html.lang};
      const noindex=[...document.querySelectorAll('meta[name="robots"]')].map(m=>m.content);
      return {title:document.title,overflow,enabledCommerce:enabledCommerce.map(x=>x.outerHTML.slice(0,180)),scripts,focusableCount:focusables.length,directions,noindex,forms:document.forms.length};
    });
    report.push({target,check,pass:check.overflow===0 && check.enabledCommerce.length===0 && check.scripts.length===0 && check.forms===0 && (target.width!==390 || check.focusableCount>0)});
  }
  const arabic=report.find(x=>x.target.name==='arabic_product');
  if (arabic) arabic.pass = arabic.pass && arabic.check.directions.html==='rtl' && arabic.check.directions.lang==='ar';
  return {passed:report.filter(x=>x.pass).length,total:report.length,report};
}
